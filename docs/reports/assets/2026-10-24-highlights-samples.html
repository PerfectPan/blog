import { createHighlighterCore } from 'shiki/core';
import type { LanguageInput } from 'shiki/core';

type CacheEntry = { html: string; expiresAt: number };

const TTL_MS = 60_000;
const cache = new Map<string, CacheEntry>();

export async function highlight(
  code: string,
  lang: string,
): Promise<string> {
  const key = `${lang}:${code.length}`;
  const hit = cache.get(key);
  if (hit && hit.expiresAt > Date.now()) {
    return hit.html;
  }
  // Reuse the engine and loaded grammars across requests.
  const highlighter = await getHighlighter(lang);
  const html = highlighter.codeToHtml(code, {
    lang,
    themes: { light: 'vitesse-light', dark: 'vitesse-dark' },
  });
  cache.set(key, { html, expiresAt: Date.now() + TTL_MS });
  return html;
}

async function getHighlighter(lang: string) {
  const mod = (await import(`shiki/langs/${lang}.mjs`)) as {
    default: LanguageInput;
  };
  return createHighlighterCore({ themes: [], langs: [mod.default] });
}

interface Stats {
  total: number;
  cached: number;
  p50: number;
  p99: number;
}

export function summarize(samples: number[]): Stats {
  const sorted = [...samples].sort((a, b) => a - b);
  const at = (p: number) => sorted[Math.min(sorted.length - 1, Math.floor(p * sorted.length))] ?? 0;
  return { total: sorted.length, cached: 0, p50: at(0.5), p99: at(0.99) };
}

import { useEffect, useRef, useState } from 'react';
import { Copy, Check } from 'lucide-react';

type Props = {
  code: string;
  lang: string;
  onCopied?: () => void;
};

export function CodeBlock({ code, lang, onCopied }: Props) {
  const [copied, setCopied] = useState(false);
  const timer = useRef<ReturnType<typeof setTimeout>>(undefined);

  useEffect(() => {
    return () => clearTimeout(timer.current);
  }, []);

  async function handleCopy() {
    try {
      await navigator.clipboard.writeText(code);
      setCopied(true);
      onCopied?.();
      timer.current = setTimeout(() => setCopied(false), 1500);
    } catch {
      // Clipboard unavailable — no-op.
    }
  }

  return (
    <div className="group relative">
      <button
        type="button"
        aria-label="Copy code"
        onClick={handleCopy}
        className="absolute top-2 right-2 opacity-0 group-hover:opacity-100"
      >
        {copied ? <Check size={12} /> : <Copy size={12} />}
      </button>
      <pre className="overflow-x-auto">
        <code className={`language-${lang}`}>{code}</code>
      </pre>
    </div>
  );
}

-- Post visibility is enforced in the data layer, not just the loader.
CREATE TABLE post (
  id         TEXT PRIMARY KEY,
  slug       TEXT NOT NULL UNIQUE,
  title      TEXT NOT NULL,
  content    TEXT NOT NULL,
  visibility TEXT NOT NULL DEFAULT 'public'
              CHECK (visibility IN ('public','member','vip','admin','password')),
  published_at INTEGER,
  created_at  INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE INDEX idx_post_visibility_published
  ON post(visibility, published_at DESC);

WITH visible AS (
  SELECT p.*, u.name AS author
  FROM post p
  JOIN user u ON u.id = p.author_id
  WHERE p.visibility = 'public'
     OR (p.visibility = 'member' AND u.role != 'guest')
  ORDER BY p.published_at DESC
  LIMIT 20 OFFSET 0
)
SELECT slug, title, author, published_at
FROM visible
WHERE published_at <= unixepoch()
GROUP BY slug;

UPDATE post
SET visibility = 'member'
WHERE visibility = 'public'
  AND published_at < unixepoch() - 86400 * 30;

EXPLAIN QUERY PLAN
SELECT count(*) FROM post WHERE visibility = 'public';

/**
 * Repository unit 1: manages record lifecycle for table_1.
 * Generated section follows the same data-access pattern as the rest.
 */
export interface Row1 {
  id: string;
  name: string;
  kind: "alpha";
  tags: string[];
  score: number;
  createdAt: number;
  updatedAt: number | null;
}

const FIELDS_1 = ["id", "name", "kind", "tags", "score", "created_at", "updated_at"] as const;

export async function list1(db: D1Database, opts: { limit?: number; offset?: number } = {}) {
  const limit = Math.min(opts.limit ?? 20, 100);
  const offset = Math.max(opts.offset ?? 0, 0);
  const { results } = await db
    .prepare(`SELECT ${FIELDS_1.join(", ")} FROM table_1 ORDER BY created_at DESC LIMIT ? OFFSET ?`)
    .bind(limit, offset)
    .all<Row1>();
  return results.map((row) => ({ ...row, tags: JSON.parse(row.tags ?? "[]") }));
}

export async function upsert1(db: D1Database, row: Partial<Row1>) {
  if (!row.id) throw new Error("upsert1: id required");
  const now = Date.now();
  await db
    .prepare(
      `INSERT INTO table_1 (id, name, kind, tags, score, created_at, updated_at)
       VALUES (?, ?, ?, ?, ?, ?, ?)
       ON CONFLICT(id) DO UPDATE SET name = excluded.name, updated_at = excluded.updated_at`,
    )
    .bind(row.id, row.name ?? "", row.kind ?? "alpha", JSON.stringify(row.tags ?? []), row.score ?? 0, now, now)
    .run();
  return { ...row, createdAt: now, updatedAt: now } as Row1;
}

检查项结论
输出 HTML 形态根元素同为 <pre style="background-color:#ffffff;--shiki-dark-bg:#121212;color:#393a34;--shiki-dark:#dbd7caee"><code>，仅 class 不同（highlights vs shiki shiki-themes vitesse-light vitesse-dark）；span 为内联 color + --shiki-dark 变量，e2e 断言 span[style*="color"] 与 data-highlighted 均满足。pierre 无 <span class="line"> 行包装（span 数少 8-20%，博客 CSS 无 .line 依赖）；无 tabindex（upgrade 不复制该属性，无影响）。
暗色切换单 HTML 双主题，与现状同构：span 内联 light 色 + --shiki-dark 变量。实测（真实浏览器 computed style）：切换 html.dark 后 A/B 两列首 span 颜色完全一致（rgb(77,147,117)）、pre 背景同为 #0a1016。styles.css 零改动。备选：defaultColor:false（全走变量，需补 light 变量 CSS）、defaultColor:'light-dark()'（CSS light-dark()，零切换 CSS）。
复制按钮upgradeCodeBlock 只换 code.innerHTML，pre.textContent 在 A/B 输出下均与源码逐字节一致（浏览器实测 1346/17724 字符全等）。复制零改动。
语言别名现状 17 语言全支持；常用别名全部内置（大小写不敏感），LANG_ALIASES 可删。 LANG_IMPORTS 懒加载映射也可删（单 WASM 含全部 lexer，import 后即时可用）。
